'''Tenant- and channel-scoped SQLAlchemy operations for minimal chat storage.'''

from typing import Any
from uuid import UUID, uuid4

from sqlalchemy import and_, exists, func, insert, select, update
from sqlalchemy.dialects.postgresql import insert as pg_insert
from sqlalchemy.ext.asyncio import AsyncConnection
from sqlalchemy.sql import Executable

from app.chat.entities import ChannelAccount, ChannelUser, Conversation, Message, Tenant

Row = dict[str, Any]
TENANTS = Tenant.__table__
CHANNELS = ChannelAccount.__table__
USERS = ChannelUser.__table__
CONVERSATIONS = Conversation.__table__
MESSAGES = Message.__table__
PLATFORMS = frozenset(('TELEGRAM', 'WHATSAPP'))


async def register_telegram_channel(
    connection: AsyncConnection,
    tenant_id: UUID,
    external_bot_id: int,
    username: str | None,
) -> UUID:
    '''Provision the local tenant and bot without changing existing ownership.'''
    if not isinstance(tenant_id, UUID) or external_bot_id <= 0:
        raise ValueError('TELEGRAM_CHANNEL_INVALID')
    external_account_id = str(external_bot_id)
    await connection.execute(
        pg_insert(TENANTS).values(id=tenant_id).on_conflict_do_nothing(
            index_elements=[TENANTS.c.id]
        )
    )
    await connection.execute(
        pg_insert(CHANNELS).values(
            id=uuid4(),
            tenant_id=tenant_id,
            platform='TELEGRAM',
            external_account_id=external_account_id,
            username=username,
        ).on_conflict_do_nothing(
            index_elements=[CHANNELS.c.platform, CHANNELS.c.external_account_id]
        )
    )
    channel_id = await connection.scalar(
        select(CHANNELS.c.id).where(
            CHANNELS.c.tenant_id == tenant_id,
            CHANNELS.c.platform == 'TELEGRAM',
            CHANNELS.c.external_account_id == external_account_id,
        )
    )
    if channel_id is None:
        raise RuntimeError('CHANNEL_ACCOUNT_UNAVAILABLE')
    await connection.execute(
        update(CHANNELS).where(
            CHANNELS.c.id == channel_id,
            CHANNELS.c.tenant_id == tenant_id,
            CHANNELS.c.platform == 'TELEGRAM',
            CHANNELS.c.external_account_id == external_account_id,
        ).values(username=username)
    )
    return channel_id


class Repository:
    def __init__(self, tenant_id: UUID, channel_id: UUID, platform: str) -> None:
        if not isinstance(tenant_id, UUID) or not isinstance(channel_id, UUID):
            raise ValueError('REPOSITORY_SCOPE_INVALID')
        if platform not in PLATFORMS:
            raise ValueError('REPOSITORY_PLATFORM_INVALID')
        self.tenant_id = tenant_id
        self.channel_id = channel_id
        self.platform = platform

    async def _first(self, connection: AsyncConnection, statement: Executable) -> Row | None:
        result = await connection.execute(statement)
        mapping = result.mappings().first()
        return dict(mapping) if mapping is not None else None

    def _channel_exists(self):
        return exists(
            select(CHANNELS.c.id).where(
                CHANNELS.c.id == self.channel_id,
                CHANNELS.c.tenant_id == self.tenant_id,
                CHANNELS.c.platform == self.platform,
            )
        )

    def _owned_conversation(self, user_id: UUID, external_chat_id: str):
        return and_(
            self._channel_exists(),
            CONVERSATIONS.c.channel_id == self.channel_id,
            CONVERSATIONS.c.platform == self.platform,
            CONVERSATIONS.c.user_id == user_id,
            CONVERSATIONS.c.external_chat_id == external_chat_id,
        )

    async def upsert_user(
        self,
        connection: AsyncConnection,
        external_user_id: str,
        username: str | None,
        first_name: str | None,
    ) -> UUID:
        if not external_user_id or len(external_user_id) > 128:
            raise ValueError('EXTERNAL_USER_ID_INVALID')
        if not await connection.scalar(select(self._channel_exists())):
            raise RuntimeError('CHANNEL_ACCOUNT_UNAVAILABLE')
        statement = pg_insert(USERS).values(
            id=uuid4(),
            platform=self.platform,
            external_user_id=external_user_id,
            username=username,
            first_name=first_name,
        )
        user_id = await connection.scalar(
            statement.on_conflict_do_update(
                index_elements=[USERS.c.platform, USERS.c.external_user_id],
                set_={
                    'username': statement.excluded.username,
                    'first_name': statement.excluded.first_name,
                },
            ).returning(USERS.c.id)
        )
        if user_id is None:
            raise RuntimeError('USER_NOT_RETURNED')
        return user_id

    async def find_duplicate_incoming(
        self,
        connection: AsyncConnection,
        user_id: UUID,
        external_chat_id: str,
        external_message_id: str,
    ) -> bool:
        '''Find a USER provider ID across all sessions of this owner and chat.'''
        if not external_message_id or len(external_message_id) > 512:
            raise ValueError('EXTERNAL_MESSAGE_ID_INVALID')
        found = await connection.scalar(
            select(MESSAGES.c.id).select_from(
                MESSAGES.join(
                    CONVERSATIONS,
                    MESSAGES.c.conversation_id == CONVERSATIONS.c.id,
                ).join(
                    CHANNELS,
                    and_(
                        CONVERSATIONS.c.channel_id == CHANNELS.c.id,
                        CONVERSATIONS.c.platform == CHANNELS.c.platform,
                    ),
                )
            ).where(
                CHANNELS.c.tenant_id == self.tenant_id,
                CHANNELS.c.id == self.channel_id,
                CHANNELS.c.platform == self.platform,
                CONVERSATIONS.c.user_id == user_id,
                CONVERSATIONS.c.external_chat_id == external_chat_id,
                MESSAGES.c.role == 'USER',
                MESSAGES.c.external_message_id == external_message_id,
            ).limit(1)
        )
        return found is not None

    async def resolve_conversation(
        self,
        connection: AsyncConnection,
        user_id: UUID,
        external_chat_id: str,
        reset: bool = False,
    ) -> UUID:
        if not isinstance(user_id, UUID) or not external_chat_id or len(external_chat_id) > 128:
            raise ValueError('CONVERSATION_OWNER_INVALID')
        if not await connection.scalar(select(self._channel_exists())):
            raise RuntimeError('CHANNEL_ACCOUNT_UNAVAILABLE')
        if reset:
            await connection.execute(
                update(CONVERSATIONS).where(
                    self._owned_conversation(user_id, external_chat_id),
                    CONVERSATIONS.c.status == 'ACTIVE',
                ).values(status='CLOSED')
            )
        conversation_id = await connection.scalar(
            select(CONVERSATIONS.c.id).where(
                self._owned_conversation(user_id, external_chat_id),
                CONVERSATIONS.c.status == 'ACTIVE',
            ).limit(1)
        )
        if conversation_id is not None:
            return conversation_id
        conversation_id = await connection.scalar(
            insert(CONVERSATIONS).values(
                id=uuid4(),
                channel_id=self.channel_id,
                platform=self.platform,
                user_id=user_id,
                external_chat_id=external_chat_id,
                status='ACTIVE',
            ).returning(CONVERSATIONS.c.id)
        )
        if conversation_id is None:
            raise RuntimeError('CONVERSATION_NOT_RETURNED')
        return conversation_id

    async def insert_message(
        self,
        connection: AsyncConnection,
        conversation_id: UUID,
        user_id: UUID,
        external_chat_id: str,
        role: str,
        content: str,
        external_message_id: str | None = None,
    ) -> int:
        if role not in ('USER', 'ASSISTANT') or not content:
            raise ValueError('MESSAGE_INVALID')
        if role == 'USER' and (not external_message_id or len(external_message_id) > 512):
            raise ValueError('EXTERNAL_MESSAGE_ID_INVALID')
        if role == 'ASSISTANT' and external_message_id is not None:
            raise ValueError('ASSISTANT_EXTERNAL_ID_INVALID')
        owned = await connection.scalar(
            select(CONVERSATIONS.c.id).where(
                CONVERSATIONS.c.id == conversation_id,
                self._owned_conversation(user_id, external_chat_id),
                CONVERSATIONS.c.status == 'ACTIVE',
            )
        )
        if owned is None:
            raise RuntimeError('CONVERSATION_SCOPE_MISMATCH')
        message_id = await connection.scalar(
            insert(MESSAGES).values(
                conversation_id=conversation_id,
                external_message_id=external_message_id,
                role=role,
                content=content,
            ).returning(MESSAGES.c.id)
        )
        if message_id is None:
            raise RuntimeError('MESSAGE_NOT_RETURNED')
        result = await connection.execute(
            update(CONVERSATIONS).where(
                CONVERSATIONS.c.id == conversation_id,
                self._owned_conversation(user_id, external_chat_id),
                CONVERSATIONS.c.status == 'ACTIVE',
            ).values(last_message_at=func.now())
        )
        if result.rowcount != 1:
            raise RuntimeError('CONVERSATION_SCOPE_MISMATCH')
        return int(message_id)

    async def completed_history(
        self,
        connection: AsyncConnection,
        conversation_id: UUID,
        user_id: UUID,
        external_chat_id: str,
        before_message_id: int,
        limit: int = 64,
    ) -> list[tuple[str, str]]:
        '''Pair recent ordered roles; an unmatched USER is replaced by the next USER.'''
        if not 1 <= limit <= 64:
            raise ValueError('HISTORY_LIMIT_INVALID')
        result = await connection.execute(
            select(MESSAGES.c.role, MESSAGES.c.content).select_from(
                MESSAGES.join(
                    CONVERSATIONS,
                    MESSAGES.c.conversation_id == CONVERSATIONS.c.id,
                ).join(
                    CHANNELS,
                    and_(
                        CONVERSATIONS.c.channel_id == CHANNELS.c.id,
                        CONVERSATIONS.c.platform == CHANNELS.c.platform,
                    ),
                )
            ).where(
                CHANNELS.c.tenant_id == self.tenant_id,
                CHANNELS.c.id == self.channel_id,
                CHANNELS.c.platform == self.platform,
                CONVERSATIONS.c.id == conversation_id,
                CONVERSATIONS.c.user_id == user_id,
                CONVERSATIONS.c.external_chat_id == external_chat_id,
                MESSAGES.c.id < before_message_id,
            ).order_by(MESSAGES.c.id.desc()).limit(limit)
        )
        pending_user: str | None = None
        pairs: list[tuple[str, str]] = []
        for role, content in reversed(result.all()):
            if role == 'USER':
                pending_user = content
            elif role == 'ASSISTANT' and pending_user is not None:
                pairs.append((pending_user, content))
                pending_user = None
        return pairs

'''Serial chat use cases with short database transactions.'''

import asyncio

from app.ai.openrouter import AIError, OpenRouterClient
from app.ai.prompts import MAX_INPUT_CHARS, build_prompt
from app.chat.repositories import Repository
from app.core.database import Database


class ChatService:
    def __init__(
        self,
        database: Database,
        repository: Repository,
        ai: OpenRouterClient,
    ) -> None:
        self.database = database
        self.repository = repository
        self.ai = ai
        self._serial = asyncio.Lock()

    async def handle(
        self,
        *,
        external_user_id: str,
        external_chat_id: str,
        external_message_id: str,
        username: str | None,
        first_name: str | None,
        kind: str,
        content: str,
    ) -> str | None:
        '''Handle one permitted SDK message; None means a duplicate TEXT was stored.'''
        if kind not in ('TEXT', 'START', 'NEW', 'SESSION', 'INVALID', 'UNSUPPORTED'):
            raise ValueError('MESSAGE_KIND_INVALID')
        if not external_chat_id or len(external_chat_id) > 128:
            raise ValueError('EXTERNAL_CHAT_ID_INVALID')
        if not external_message_id or len(external_message_id) > 512:
            raise ValueError('EXTERNAL_MESSAGE_ID_INVALID')
        if kind == 'TEXT' and (not content.strip() or len(content) > MAX_INPUT_CHARS):
            kind = 'INVALID'
        display_name = (first_name or username or external_user_id).strip()
        if not display_name:
            display_name = external_user_id
        async with self._serial:
            async with self.database.transaction() as connection:
                user_id = await self.repository.upsert_user(
                    connection, external_user_id, username, first_name
                )
                if kind == 'TEXT':
                    if await self.repository.find_duplicate_incoming(
                        connection, user_id, external_chat_id, external_message_id
                    ):
                        return None
                if kind in ('INVALID', 'UNSUPPORTED'):
                    return self._command_answer(kind, display_name, None)
                conversation_id = await self.repository.resolve_conversation(
                    connection, user_id, external_chat_id, reset=kind == 'NEW'
                )
                if kind != 'TEXT':
                    return self._command_answer(kind, display_name, conversation_id)
                current_id = await self.repository.insert_message(
                    connection,
                    conversation_id,
                    user_id,
                    external_chat_id,
                    'USER',
                    content,
                    external_message_id,
                )
                pairs = await self.repository.completed_history(
                    connection, conversation_id, user_id, external_chat_id, current_id
                )
                prompt = build_prompt(display_name, username, pairs, content)
            completion = None
            for attempt in range(1, 4):
                try:
                    completion = await self.ai.reply(prompt)
                    break
                except AIError as error:
                    if not error.retryable or attempt == 3:
                        return self._error_message(error.code)
                    await asyncio.sleep(max(error.delay, attempt * 5))
            if completion is None:
                raise RuntimeError('AI_COMPLETION_MISSING')
            async with self.database.transaction() as connection:
                await self.repository.insert_message(
                    connection,
                    conversation_id,
                    user_id,
                    external_chat_id,
                    'ASSISTANT',
                    completion.content,
                )
            return completion.content

    def _command_answer(self, kind: str, name: str, session_id: object) -> str:
        if kind == 'START':
            return f'Halo, {name}! Silakan kirim pertanyaan. Sesi: {session_id}'
        if kind == 'NEW':
            return f'Sesi baru dibuat, {name}. Sesi: {session_id}'
        if kind == 'SESSION':
            return f'{name}, sesi aktif Anda: {session_id}'
        if kind == 'INVALID':
            return 'Kirim pesan teks berisi 1 sampai 4000 karakter.'
        return 'Perintah tersedia: /start, /new, /session.'

    def _error_message(self, code: str) -> str:
        if code == 'AI_NOT_CONFIGURED':
            return 'AI belum dikonfigurasi. Operator perlu mengisi OPENROUTER_API_KEY dan AI_MODEL.'
        if code in ('AI_HTTP_401', 'AI_HTTP_403'):
            return 'Akses AI ditolak. Hubungi operator untuk memeriksa konfigurasi.'
        if code == 'AI_HTTP_402':
            return 'Saldo layanan AI tidak mencukupi. Hubungi operator.'
        if code == 'AI_TRANSPORT_UNKNOWN':
            return (
                'Koneksi AI terputus; hasil permintaan tidak diketahui. '
                'Kirim pertanyaan baru bila perlu.'
            )
        return 'Layanan AI belum dapat menjawab. Silakan kirim pertanyaan baru nanti.'

class RepositoryTests(unittest.IsolatedAsyncioTestCase):
    def setUp(self) -> None:
        self.repository = Repository("unit-scope", 71)
        self.connection = Mock(spec=AsyncConnection)
        self.connection.execute = AsyncMock()
        self.connection.scalar = AsyncMock(return_value=3)
        self.inbox: dict[str, Any] = {
            "id": uuid4(), "identity_id": uuid4(), "chat_id": 7, "update_id": 81
        }

    def result(self, row: dict[str, Any] | None) -> Mock:
        result = Mock()
        result.mappings.return_value.first.return_value = row
        return result

    def statement(self, index: int) -> tuple[str, dict[str, Any]]:
        call = self.connection.execute.await_args_list[index]
        self.assertEqual(len(call.args), 1)
        self.assertFalse(call.kwargs)
        return self.compiled(call.args[0])

    def compiled(self, statement: Executable) -> tuple[str, dict[str, Any]]:
        self.assertNotIsInstance(statement, TextClause)
        compiled = statement.compile(
            dialect=postgresql.dialect(),
            compile_kwargs={"render_postcompile": True},
        )
        return re.sub(r"\s+", " ", str(compiled)).strip(), dict(compiled.params)

    def assert_scoped(self, sql: str, parameters: dict[str, Any]) -> None:
        self.assertRegex(sql, r"\bscope\s*=")
        self.assertRegex(sql, r"\bbot_id\s*=")
        self.assertIn("unit-scope", parameters.values())
        self.assertIn(71, parameters.values())

class TransactionMarkers:
    def __init__(self, connection: Mock) -> None:
        self.connection = connection
        self.active = False
        self.number = 0
        self.events: list[tuple[str, int]] = []

    @asynccontextmanager
    async def transaction(self) -> AsyncIterator[Mock]:
        if self.active:
            raise AssertionError("Nested transaction")
        self.number += 1
        number = self.number
        self.active = True
        self.events.append(("begin", number))
        try:
            yield self.connection
        except BaseException:
            self.events.append(("rollback", number))
            raise
        else:
            self.events.append(("commit", number))
        finally:
            self.active = False
