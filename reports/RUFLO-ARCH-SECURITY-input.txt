## Application structure

One Java 17 / Spring Boot 4.1.1 backend: a modular monolith organized in existing top-level packages under `org.lite.project.cashiersaas`.

```mermaid
flowchart LR
    HTTP[Staff or public HTTP client] --> Security[Spring Security / JWT]
    Security --> Controller[controller]
    Controller --> Service[service: use cases and transactions]
    Service --> Guard[MerchantAccessGuard]
    Service --> Repository[repository: JPA / Blaze-Persistence]
    Repository --> DB[(PostgreSQL)]
    Service --> Mapper[dto: explicit mappers]
    Mapper --> Response[ApiResponse and named DTOs]
```

Controllers adapt HTTP and return `ApiResponse<T>`; services coordinate transactions, guards, persistence and explicit DTO mappers. Entities own named business transitions. Repositories use EntityManager and Blaze-Persistence; QueryDSL generated Q types support typed paths. `ApiResponse` omits null fields. Current exception classes reside in `exception/runtime`, with `GlobalExceptionHandler`; AGENTS.md instead prescribes `shared/exception/runtime` for the policy boundary. Account scope supports onboarding and merchant selection; merchant scope carries tenant/outlet/membership context. Refresh tokens are persisted as digests and rotated; sessions support revocation. ## Pay First state and consistency

```text
QR/POS checkout -> NEW + UNPAID
successful settlement -> CONFIRMED + PAID
kitchen -> PREPARING -> READY -> COMPLETED
NEW + UNPAID -> CANCELLED (reason and audit in service)
```

SalesOrderEntity enforces these transitions. Discount and cancellation require NEW/UNPAID. Pricing uses BigDecimal, server-side availability/pricing and order snapshots. package org.lite.project.cashiersaas.configuration;

import com.nimbusds.jose.jwk.source.ImmutableSecret;
import org.lite.project.cashiersaas.component.AudienceValidator;
import org.lite.project.cashiersaas.component.TenantJwtAuthenticationConverter;
import org.springframework.boot.context.properties.EnableConfigurationProperties;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import org.springframework.core.convert.converter.Converter;
import org.springframework.security.authentication.AbstractAuthenticationToken;
import org.springframework.security.oauth2.core.DelegatingOAuth2TokenValidator;
import org.springframework.security.oauth2.core.OAuth2TokenValidator;
import org.springframework.security.oauth2.core.OAuth2TokenValidatorResult;
import org.springframework.security.oauth2.core.OAuth2Error;
import org.springframework.security.oauth2.jose.jws.MacAlgorithm;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.security.oauth2.jwt.JwtDecoder;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.JwtValidators;
import org.springframework.security.oauth2.jwt.NimbusJwtDecoder;
import org.springframework.security.oauth2.jwt.NimbusJwtEncoder;

import javax.crypto.SecretKey;
import javax.crypto.spec.SecretKeySpec;
import java.util.Base64;

@Configuration
@EnableConfigurationProperties(JwtProperties.class)
public class JwtConfiguration {

    private static final int MIN_HMAC_SHA256_KEY_BYTES = 32;

    @Bean
    public SecretKey jwtSigningKey(JwtProperties jwtProperties) {
        String configuredSecret = jwtProperties.getSecret();
        if (configuredSecret == null || configuredSecret.isBlank()) {
            throw new IllegalStateException("CASHIER_JWT_SECRET_B64 must be configured.");
        }

        byte[] secretBytes;
        try {
            secretBytes = Base64.getDecoder().decode(configuredSecret);
        } catch (IllegalArgumentException exception) {
            throw new IllegalStateException("CASHIER_JWT_SECRET_B64 must be Base64-encoded.", exception);
        }

        if (secretBytes.length < MIN_HMAC_SHA256_KEY_BYTES) {
            throw new IllegalStateException("CASHIER_JWT_SECRET_B64 must decode to at least 32 bytes.");
        }

        return new SecretKeySpec(secretBytes, "HmacSHA256");
    }

    @Bean
    public JwtEncoder jwtEncoder(SecretKey jwtSigningKey) {
        return new NimbusJwtEncoder(new ImmutableSecret<>(jwtSigningKey));
    }

    @Bean
    @Primary
    public JwtDecoder jwtDecoder(SecretKey jwtSigningKey, JwtProperties jwtProperties) {
        NimbusJwtDecoder decoder = NimbusJwtDecoder.withSecretKey(jwtSigningKey).macAlgorithm(MacAlgorithm.HS256).build();
        OAuth2TokenValidator<Jwt> requiredClaims = jwt -> {
            if (jwt.getExpiresAt() == null || jwt.getIssuedAt() == null || jwt.getId() == null
                    || jwt.getId().isBlank() || jwt.getSubject() == null || jwt.getSubject().isBlank()) {
                return OAuth2TokenValidatorResult.failure(new OAuth2Error("invalid_token",
                        "Required JWT claims are missing.", null));
            }
            return OAuth2TokenValidatorResult.success();
        };
        OAuth2TokenValidator<Jwt> validator = new DelegatingOAuth2TokenValidator<>(
                JwtValidators.createDefaultWithIssuer(jwtProperties.getIssuer()),
                new AudienceValidator(jwtProperties.getAudience()), requiredClaims);
        decoder.setJwtValidator(validator);
        return decoder;
    }

    @Bean
    public Converter<Jwt, AbstractAuthenticationToken> jwtAuthenticationConverter() {
        return new TenantJwtAuthenticationConverter();
    }
}

package org.lite.project.cashiersaas.component;

import org.lite.project.cashiersaas.enums.Role;
import org.springframework.core.convert.converter.Converter;
import org.springframework.security.authentication.BadCredentialsException;
import org.springframework.security.authentication.AbstractAuthenticationToken;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.oauth2.jwt.Jwt;

import java.util.LinkedHashSet;
import java.util.List;
import java.util.Set;
import java.util.UUID;

public class TenantJwtAuthenticationConverter implements Converter<Jwt, AbstractAuthenticationToken> {

    @Override
    public AbstractAuthenticationToken convert(Jwt jwt) {
        UUID userId = requiredUuidClaim(jwt, "sub");
        UUID sessionId = requiredUuidClaim(jwt, "session_id");
        if ("ACCOUNT".equals(jwt.getClaimAsString("scope"))) {
            AccountPrincipal principal = new AccountPrincipal(userId, requiredEmail(jwt), sessionId);
            Set<SimpleGrantedAuthority> authorities = Set.of(new SimpleGrantedAuthority("SCOPE_ACCOUNT"));
            return new UsernamePasswordAuthenticationToken(principal, null, authorities);
        }

        if (!"MERCHANT".equals(jwt.getClaimAsString("scope"))) {
            throw new BadCredentialsException("JWT scope is invalid.");
        }

        UUID tenantId = requiredUuidClaim(jwt, "tenant_id");
        UUID outletId = requiredUuidClaim(jwt, "outlet_id");
        UUID membershipId = requiredUuidClaim(jwt, "membership_id");
        Set<Role> roles = requiredRoles(jwt);

        TenantPrincipal principal = new TenantPrincipal(userId, tenantId, outletId, membershipId, sessionId, roles);
        Set<SimpleGrantedAuthority> authorities = roles.stream()
                .map(role -> new SimpleGrantedAuthority("ROLE_" + role.name()))
                .collect(java.util.stream.Collectors.toUnmodifiableSet());

        return new UsernamePasswordAuthenticationToken(principal, null, authorities);
    }

    private String requiredEmail(Jwt jwt) {
        String email = jwt.getClaimAsString("email");
        if (email == null || email.isBlank()) {
            throw new BadCredentialsException("JWT claim 'email' is required.");
        }
        return email;
    }

    private UUID requiredUuidClaim(Jwt jwt, String claimName) {
        String claimValue = jwt.getClaimAsString(claimName);
        if (claimValue == null || claimValue.isBlank()) {
            throw new BadCredentialsException("JWT claim '" + claimName + "' must be a UUID.");
        }

        try {
            return UUID.fromString(claimValue);
        } catch (IllegalArgumentException exception) {
            throw new BadCredentialsException("JWT claim '" + claimName + "' must be a UUID.", exception);
        }
    }

    private Set<Role> requiredRoles(Jwt jwt) {
        List<String> roleClaims = jwt.getClaimAsStringList("roles");
        if (roleClaims == null || roleClaims.isEmpty()) {
            throw new BadCredentialsException("JWT claim 'roles' is required.");
        }

        try {
            return roleClaims.stream()
                    .map(Role::valueOf)
                    .collect(java.util.stream.Collectors.toCollection(LinkedHashSet::new));
        } catch (IllegalArgumentException exception) {
            throw new BadCredentialsException("JWT contains an unsupported role.", exception);
        }
    }
}

package org.lite.project.cashiersaas.component;

import lombok.RequiredArgsConstructor;
import org.lite.project.cashiersaas.configuration.JwtProperties;
import org.lite.project.cashiersaas.enums.Role;
import org.springframework.security.oauth2.jose.jws.MacAlgorithm;
import org.springframework.security.oauth2.jwt.JwsHeader;
import org.springframework.security.oauth2.jwt.JwtClaimsSet;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.JwtEncoderParameters;
import org.springframework.stereotype.Service;

import java.time.Instant;
import java.util.List;
import java.util.Objects;
import java.util.UUID;

@Service
@RequiredArgsConstructor
public class JwtTokenService {

    private final JwtEncoder jwtEncoder;
    private final JwtProperties jwtProperties;

    public String generateAccessToken(MerchantTokenContext context) {
        requireTokenSubject(context);

        Instant issuedAt = Instant.now();
        List<String> roleNames = context.getRoles().stream().map(Role::name).distinct().toList();
        JwtClaimsSet claims = JwtClaimsSet.builder()
                .issuer(jwtProperties.getIssuer())
                .audience(List.of(jwtProperties.getAudience()))
                .subject(context.getUserId().toString())
                .issuedAt(issuedAt)
                .expiresAt(context.getExpiresAt())
                .id(UUID.randomUUID().toString())
                .claim("tenant_id", context.getTenantId().toString())
                .claim("outlet_id", context.getOutletId().toString())
                .claim("membership_id", context.getMembershipId().toString())
                .claim("session_id", context.getSessionId().toString())
                .claim("scope", "MERCHANT")
                .claim("email", context.getEmail())
                .claim("roles", roleNames)
                .build();

        JwsHeader header = JwsHeader.with(MacAlgorithm.HS256)
                .type("JWT")
                .build();

        return jwtEncoder.encode(JwtEncoderParameters.from(header, claims)).getTokenValue();
    }

    public String generateAccountAccessToken(UUID userId, String email, UUID sessionId) {
        return generateAccountAccessToken(userId, email, sessionId,
                Instant.now().plus(jwtProperties.getAccessTokenTtl()));
    }

    public String generateAccountAccessToken(UUID userId, String email, UUID sessionId, Instant expiresAt) {
        Objects.requireNonNull(userId, "userId is required.");
        Objects.requireNonNull(email, "email is required.");
        Objects.requireNonNull(sessionId, "sessionId is required.");
        Objects.requireNonNull(expiresAt, "expiresAt is required.");
        if (email.isBlank()) {
            throw new IllegalArgumentException("email is required.");
        }

        Instant issuedAt = Instant.now();
        JwtClaimsSet claims = JwtClaimsSet.builder()
                .issuer(jwtProperties.getIssuer())
                .audience(List.of(jwtProperties.getAudience()))
                .subject(userId.toString())
                .issuedAt(issuedAt)
                .expiresAt(expiresAt)
                .id(UUID.randomUUID().toString())
                .claim("email", email)
                .claim("scope", "ACCOUNT")
                .claim("session_id", sessionId.toString())
                .build();

        JwsHeader header = JwsHeader.with(MacAlgorithm.HS256)
                .type("JWT")
                .build();

        return jwtEncoder.encode(JwtEncoderParameters.from(header, claims)).getTokenValue();
    }

    private void requireTokenSubject(MerchantTokenContext context) {
        Objects.requireNonNull(context, "context is required.");
        Objects.requireNonNull(context.getUserId(), "userId is required.");
        Objects.requireNonNull(context.getTenantId(), "tenantId is required.");
        Objects.requireNonNull(context.getOutletId(), "outletId is required.");
        Objects.requireNonNull(context.getMembershipId(), "membershipId is required.");
        Objects.requireNonNull(context.getSessionId(), "sessionId is required.");
        Objects.requireNonNull(context.getEmail(), "email is required.");
        Objects.requireNonNull(context.getExpiresAt(), "expiresAt is required.");

        if (context.getEmail().isBlank()) {
            throw new IllegalArgumentException("email is required.");
        }
        if (context.getRoles() == null || context.getRoles().isEmpty()) {
            throw new IllegalArgumentException("At least one role is required.");
        }
    }
}

@Service
@RequiredArgsConstructor
public class AuthSessionService {
    private static final SecureRandom RANDOM = new SecureRandom();

    private final AuthSessionRepository sessionRepository;
    private final AuthRefreshTokenRepository tokenRepository;
    private final TenantMembershipRepository membershipRepository;
    private final SubscriptionRepository subscriptionRepository;
    private final OutletRepository outletRepository;
    private final MembershipOutletAccessRepository accessRepository;
    private final JwtTokenService jwtTokenService;
    private final JwtProperties jwtProperties;

    @Transactional
    public IssuedTokens start(StaffUserEntity user) {
        Instant now = Instant.now();
        AuthSessionEntity session = AuthSessionEntity.start(user, now.plus(jwtProperties.getRefreshTokenTtl()));
        sessionRepository.persist(session);
        String rawRefresh = newRefreshToken();
        tokenRepository.persist(AuthRefreshTokenEntity.issue(session, digest(rawRefresh), session.getExpiresAt()));
        Instant accessExpiry = min(now.plus(jwtProperties.getAccessTokenTtl()), session.getExpiresAt());
        String access = jwtTokenService.generateAccountAccessToken(user.getId(), user.getEmail(),
                session.getId(), accessExpiry);
        return new IssuedTokens(access, accessExpiry, rawRefresh,
                user.isPendingProfile() ? "COMPLETE_ONBOARDING" : "SELECT_MERCHANT_OUTLET", null, null, null);
    }


    @Transactional
    public IssuedTokens selectMerchant(AccountPrincipal principal, TenantMembershipEntity membership,
                                       OutletEntity outlet, SubscriptionEntity subscription) {
        AuthSessionEntity session = sessionRepository.findById(principal.getSessionId(), principal.getUserId())
                .orElseThrow(SessionInvalidException::new);
        sessionRepository.lockAndRefresh(session);
        if (!session.isUsableAt(Instant.now()) || !membership.getUser().getId().equals(principal.getUserId())) {
            throw new SessionInvalidException();
        }
        if (!membership.getTenant().isFnbEnabled()) {
            throw new SessionInvalidException();
        }
        session.selectMerchant(membership, outlet);
        Instant expiresAt = merchantAccessExpiry(session, subscription);
        String access = jwtTokenService.generateAccessToken(MerchantTokenContext.builder()
                .userId(principal.getUserId())
                .tenantId(membership.getTenant().getId())
                .outletId(outlet.getId())
                .membershipId(membership.getId())
                .sessionId(session.getId())
                .email(membership.getUser().getEmail())
                .roles(List.of(membership.getRole()))
                .expiresAt(expiresAt)
                .build());
        return new IssuedTokens(access, expiresAt, null, "OPERATE", membership.getTenant().getId(),
                outlet.getId(), membership.getRole());
    }

    @Transactional(noRollbackFor = SessionInvalidException.class)
    public IssuedTokens refresh(String rawRefresh) {
        AuthRefreshTokenEntity token = tokenRepository.findByDigest(digestValidated(rawRefresh))
                .orElseThrow(SessionInvalidException::new);
        AuthSessionEntity session = token.getSession();
        sessionRepository.lockAndRefresh(session);
        tokenRepository.refresh(token);
        Instant now = Instant.now();
        if (!session.isUsableAt(now) || !token.isUnusedAt(now)) {
            session.revoke(now);
            throw new SessionInvalidException();
        }
        StaffUserEntity user = session.getUser();
        IssuedTokens access = issueForExistingSession(session, user, now);
        token.consume(now);
        String newRawRefresh = newRefreshToken();
        tokenRepository.persist(AuthRefreshTokenEntity.issue(session, digest(newRawRefresh), session.getExpiresAt()));
        return new IssuedTokens(access.getAccessToken(), access.getExpiresAt(), newRawRefresh,
                access.getNextStep(), access.getTenantId(), access.getOutletId(), access.getRole());
    }

    @Transactional
    public void logout(String rawRefresh) {
        if (rawRefresh == null || rawRefresh.length() != 43) {
            return;
        }
        tokenRepository.findByDigest(digest(rawRefresh)).ifPresent(token -> {
            AuthSessionEntity session = token.getSession();
            sessionRepository.lockAndRefresh(session);
            session.revoke(Instant.now());
        });
    }

    private IssuedTokens issueForExistingSession(AuthSessionEntity session, StaffUserEntity user, Instant now) {
        if (session.getTenant() == null) {
            if (!user.isActive() && !user.isPendingProfile()) {
                session.revoke(now);
                throw new SessionInvalidException();
            }
            Instant expiry = min(now.plus(jwtProperties.getAccessTokenTtl()), session.getExpiresAt());
            String access = jwtTokenService.generateAccountAccessToken(user.getId(), user.getEmail(),
                    session.getId(), expiry);
            return new IssuedTokens(access, expiry, null,
                    user.isPendingProfile() ? "COMPLETE_ONBOARDING" : "SELECT_MERCHANT_OUTLET", null, null, null);
        }
        if (!user.isActive() || !session.getTenant().isActive() || !session.getTenant().isFnbEnabled()) {
            session.revoke(now);
            throw new SessionInvalidException();
        }
        UUID tenantId = session.getTenant().getId();
        UUID outletId = session.getOutlet().getId();
        TenantMembershipEntity membership = membershipRepository.findActive(user.getId(), tenantId)
                .filter(value -> value.getId().equals(session.getMembership().getId()))
                .filter(value -> value.getRole() == session.getRole())
                .orElse(null);
        SubscriptionEntity subscription = subscriptionRepository.findActiveForTenant(tenantId).orElse(null);
        if (membership == null || subscription == null || !subscription.isActiveAt(now)
                || outletRepository.findActive(tenantId, outletId).isEmpty()
                || accessRepository.findAccess(tenantId, membership.getId(), outletId).isEmpty()) {
            session.revoke(now);
            throw new SessionInvalidException();
        }
        Instant expiry = min(now.plus(jwtProperties.getAccessTokenTtl()), session.getExpiresAt());
        if (subscription.getEndsAt() != null) {
            expiry = min(expiry, subscription.getEndsAt());
        }
        String access = jwtTokenService.generateAccessToken(MerchantTokenContext.builder()
                .userId(user.getId())
                .tenantId(tenantId)
                .outletId(outletId)
                .membershipId(membership.getId())
                .sessionId(session.getId())
                .email(user.getEmail())
                .roles(List.of(membership.getRole()))
                .expiresAt(expiry)
                .build());
        return new IssuedTokens(access, expiry, null, "OPERATE", tenantId, outletId, membership.getRole());
    }

    private Instant merchantAccessExpiry(AuthSessionEntity session, SubscriptionEntity subscription) {
        Instant now = Instant.now();
        if (subscription == null || !subscription.isActiveAt(now)
                || !subscription.getTenant().getId().equals(session.getTenant().getId())) {
            throw new OperationConflictException(ErrorMessage.SUBSCRIPTION_INACTIVE.getMessage());
        }
        Instant expiry = min(now.plus(jwtProperties.getAccessTokenTtl()), session.getExpiresAt());
        return subscription.getEndsAt() == null ? expiry : min(expiry, subscription.getEndsAt());
    }

    private Instant min(Instant first, Instant second) {
        return first.isBefore(second) ? first : second;
    }

    private String newRefreshToken() {
        byte[] bytes = new byte[32];
        RANDOM.nextBytes(bytes);
        return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
    }

    private String digestValidated(String value) {
        if (value == null || value.length() != 43) {
            throw new SessionInvalidException();
        }
        return digest(value);
    }

    private String digest(String value) {
        try {
            return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256")
                    .digest(value.getBytes(StandardCharsets.UTF_8)));
        } catch (NoSuchAlgorithmException exception) {
            throw new IllegalStateException("SHA-256 is unavailable.", exception);
        }
    }

    @Getter
    @RequiredArgsConstructor
    public static class IssuedTokens {
        private final String accessToken;
        private final Instant expiresAt;
        private final String refreshToken;
        private final String nextStep;
        private final UUID tenantId;
        private final UUID outletId;
        private final Role role;
    }
}

package org.lite.project.cashiersaas.service;

import lombok.RequiredArgsConstructor;
import org.lite.project.cashiersaas.component.MerchantAccessContext;
import org.lite.project.cashiersaas.component.TenantPrincipal;
import org.lite.project.cashiersaas.enums.Role;
import org.lite.project.cashiersaas.entity.shared.identity.TenantMembershipEntity;
import org.lite.project.cashiersaas.exception.runtime.ResourceNotFoundException;
import org.lite.project.cashiersaas.repository.MerchantAccessRepository;
import org.springframework.stereotype.Component;

import java.time.Instant;
import java.util.UUID;

@Component
@RequiredArgsConstructor
public class MerchantAccessGuard {
    private final MerchantAccessRepository accessRepository;

    public MerchantAccessContext requireOutlet(TenantPrincipal principal, UUID outletId) {
        principal.requireOutlet(outletId);
        MerchantAccessRepository.OutletAccess access = accessRepository.findOutlet(
                        principal.getUserId(), principal.getTenantId(), outletId, principal.getRoles(), Instant.now())
                .orElseThrow(ResourceNotFoundException::new);
        TenantMembershipEntity membership = access.getMembership();
        return new MerchantAccessContext(membership.getTenant(), access.getOutlet(), membership);
    }

    public TenantMembershipEntity requireMember(TenantPrincipal principal) {
        return accessRepository.findMember(principal.getUserId(), principal.getTenantId(),
                        principal.getRoles(), Instant.now())
                .orElseThrow(ResourceNotFoundException::new);
    }

    public MerchantAccessContext requireOwner(TenantPrincipal principal, UUID outletId) {
        MerchantAccessContext context = requireOutlet(principal, outletId);
        if (context.getMembership().getRole() != Role.OWNERMERCHANT) {
            throw new ResourceNotFoundException();
        }
        return context;
    }
}

package org.lite.project.cashiersaas.repository;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import jakarta.persistence.Tuple;
import lombok.Getter;
import lombok.RequiredArgsConstructor;
import org.lite.project.cashiersaas.entity.shared.billing.SubscriptionEntity;
import org.lite.project.cashiersaas.entity.shared.identity.MembershipOutletAccessEntity;
import org.lite.project.cashiersaas.entity.shared.identity.TenantMembershipEntity;
import org.lite.project.cashiersaas.entity.shared.merchant.OutletEntity;
import org.lite.project.cashiersaas.enums.Role;
import org.lite.project.cashiersaas.shared.StatusConstants;
import org.springframework.stereotype.Repository;

import java.time.Instant;
import java.util.Optional;
import java.util.Set;
import java.util.UUID;

@Repository
@RequiredArgsConstructor
public class MerchantAccessRepository {
    private final EntityManager entityManager;
    private final CriteriaBuilderFactory criteriaBuilderFactory;

    public Optional<TenantMembershipEntity> findMember(UUID userId, UUID tenantId, Set<Role> roles, Instant now) {
        return criteriaBuilderFactory.create(entityManager, TenantMembershipEntity.class, "membership")
                .innerJoin("membership.user", "user")
                .innerJoin("membership.tenant", "tenant")
                .innerJoin("tenant.businessType", "businessType")
                .fetch("membership.tenant")
                .where("membership.user.id").eq(userId)
                .where("membership.tenant.id").eq(tenantId)
                .where("membership.active").eq(true)
                .where("membership.role").in(roles)
                .where("user.status").eq(StatusConstants.ACTIVE)
                .where("tenant.status").eq(StatusConstants.ACTIVE)
                .where("businessType.code").eq("FNB")
                .where("businessType.active").eq(true)
                .whereExists()
                    .from(SubscriptionEntity.class, "subscription")
                    .select("subscription.id")
                    .where("subscription.tenant.id").eqExpression("tenant.id")
                    .where("subscription.planCode").eq("FREETIER")
                    .where("subscription.status").eq(StatusConstants.ACTIVE)
                    .where("subscription.startsAt").le(now)
                    .whereOr()
                        .where("subscription.endsAt").isNull()
                        .where("subscription.endsAt").gt(now)
                    .endOr()
                .end()
                .setMaxResults(1).getResultList().stream().findFirst();
    }

    public Optional<OutletAccess> findOutlet(UUID userId, UUID tenantId, UUID outletId,
                                             Set<Role> roles, Instant now) {
        return criteriaBuilderFactory.create(entityManager, Tuple.class)
                .from(MembershipOutletAccessEntity.class, "access")
                .innerJoin("access.membership", "membership")
                .innerJoin("membership.user", "user")
                .innerJoin("membership.tenant", "tenant")
                .innerJoin("tenant.businessType", "businessType")
                .innerJoin("access.outlet", "outlet")
                .select("membership")
                .select("outlet")
                .fetch("membership.tenant")
                .where("access.tenant.id").eq(tenantId)
                .where("membership.tenant.id").eq(tenantId)
                .where("outlet.tenant.id").eq(tenantId)
                .where("membership.user.id").eq(userId)
                .where("outlet.id").eq(outletId)
                .where("membership.active").eq(true)
                .where("membership.role").in(roles)
                .where("user.status").eq(StatusConstants.ACTIVE)
                .where("tenant.status").eq(StatusConstants.ACTIVE)
                .where("businessType.code").eq("FNB")
                .where("businessType.active").eq(true)
                .where("outlet.active").eq(true)
                .whereExists()
                    .from(SubscriptionEntity.class, "subscription")
                    .select("subscription.id")
                    .where("subscription.tenant.id").eqExpression("tenant.id")
                    .where("subscription.planCode").eq("FREETIER")
                    .where("subscription.status").eq(StatusConstants.ACTIVE)
                    .where("subscription.startsAt").le(now)
                    .whereOr()
                        .where("subscription.endsAt").isNull()
                        .where("subscription.endsAt").gt(now)
                    .endOr()
                .end()
                .setMaxResults(1).getResultList().stream()
                .map(row -> new OutletAccess(row.get(0, TenantMembershipEntity.class),
                        row.get(1, OutletEntity.class)))
                .findFirst();
    }

    @Getter
    @RequiredArgsConstructor
    public static class OutletAccess {
        private final TenantMembershipEntity membership;
        private final OutletEntity outlet;
    }
}
